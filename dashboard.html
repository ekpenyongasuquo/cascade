"rename" in words and "to" in words: try: rename_idx = words.index("rename") to_idx = words.index("to") changed_field = words[rename_idx + 1] new_field = words[to_idx + 1] except (IndexError, ValueError): pass return { "entity_urn": entity_urn, "changed_field": changed_field, "new_field": new_field, "pr_number": pr_payload.get("number"), "pr_title": pr_payload.get("pull_request", result["valuation"]["overall_risk"], "auto_patched": result["patch"]["total_auto_patched"], "needs_review": result["patch"]["total_needs_review"], }, "incident_report":
